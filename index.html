window.uploadImageToGitHub = async function() { let pass = prompt("Masukkan Password Admin:"); if (pass !== "123") { return alert("Password salah!"); } let skuInput = document.getElementById('uploadSkuInput').value.trim().toUpperCase(); let fileInput = document.getElementById('uploadImageFile').files[0]; if (!skuInput) return alert("Masukkan Kode SKU terlebih dahulu!"); if (!fileInput) return alert("Pilih file gambar dari perangkat Anda!"); // Masukkan Token GitHub Classic yang sudah dibuat di sini let githubToken = "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"; if (githubToken.includes("MASUKKAN_TOKEN")) { return alert("Token GitHub belum dimasukkan di kodingan! Harap setel token admin terlebih dahulu."); } let reader = new FileReader(); reader.readAsDataURL(fileInput); reader.onload = async function () { let base64Content = reader.result.split(',')[1]; let fileName = `${skuInput}.jpg`; let owner = "ptgriyabajuanak-lab"; let repo = "moonaa-katalog"; let path = `assets/${fileName}`; let apiUrl = `https://api.github.com/repos/${owner}/${repo}/contents/${path}`; alert("Sedang mengunggah foto ke folder assets GitHub..."); try { // Cek dulu apakah file dengan nama SKU tersebut sudah ada sebelumnya (untuk ambil SHA-nya kalau mau di-update) let sha = null; let getResponse = await fetch(apiUrl, { headers: { "Authorization": `token ${githubToken}` } }); if (getResponse.ok) { let fileData = await getResponse.json(); sha = fileData.sha; } // Kirim data file baru ke GitHub API let bodyData = { message: `Upload foto produk SKU: ${skuInput} via Admin Web`, content: base64Content }; if (sha) bodyData.sha = sha; // Kalau file sudah ada, sertakan sha untuk menimpa file lama let putResponse = await fetch(apiUrl, { method: "PUT", headers: { "Authorization": `token ${githubToken}`, "Content-Type": "application/json" }, body: JSON.stringify(bodyData) }); if (putResponse.ok) { alert("✅ Berhasil! Foto produk berhasil di-upload langsung ke folder assets GitHub."); closeAdminUploadModal(); location.reload(); // Refresh halaman untuk menampilkan foto baru } else { let errData = await putResponse.json(); alert("Gagal upload ke GitHub: " + (errData.message || "Kesalahan tidak dikenal")); } } catch (error) { console.error(error); alert("Terjadi kesalahan jaringan saat upload."); } }; }
